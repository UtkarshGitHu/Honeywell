import collections
import unittest

def summarize_orders(events):
    get_unique_events = {}
    for event in events:
        if event['event_id'] not in get_unique_events:
            get_unique_events[event['event_id']] = event
    
    // Using list because an order can have multiple event        
    order_events = collections.defaultdict(list)
    for event in get_unique_events.values():
        order_events[event['order_id']].append(event)
        
   def get_timestamp(x)
       return x['timestamp']    
    
    // sorting by timestamp
    summaries = []
    for order_id, evs in order_events.items():
        evs.sort(key= get_timestamp))
        
        status = 'CREATED'
        items = collections.defaultdict(int)
        
        for e in evs:
            e_type = e['type']
            if e_type == 'CANCELLED':
                status = 'CANCELLED'
            elif e_type == 'PAID':
                if status != 'CANCELLED':
                    status = 'PAID'
            elif e_type == 'ITEM_ADDED':
                if status not in ('PAID', 'CANCELLED'):
                    items[e['sku']] += e.get('qty', 0)
            elif e_type == 'ITEM_REMOVED':
                if status not in ('PAID', 'CANCELLED'):
                    items[e['sku']] = max(0, items[e['sku']] - e.get('qty', 0))
                    
        final_items = {sku: qty for sku, qty in items.items() if qty > 0}
        
        summaries.append({
            'order_id': order_id,
            'status': status,
            'items': final_items
        })
        
    return sorted(summaries, key=lambda x: x['order_id'])


class TestOrderEventAggregator(unittest.TestCase):
    def test_example(self):
        events = [
            {'order_id': 'A', 'event_id': 'e2', 'type': 'ITEM_ADDED', 'sku': 'P1', 'qty': 2, 'timestamp': 2},
            {'order_id': 'A', 'event_id': 'e1', 'type': 'CREATED', 'timestamp': 1},
            {'order_id': 'A', 'event_id': 'e3', 'type': 'PAID', 'timestamp': 3},
            {'order_id': 'A', 'event_id': 'e2', 'type': 'ITEM_ADDED', 'sku': 'P1', 'qty': 2, 'timestamp': 2}
        ]
        expected = [{'order_id': 'A', 'status': 'PAID', 'items': {'P1': 2}}]
        self.assertEqual(summarize_orders(events), expected)

    def test_cancel_precedence(self):
        events = [
            {'order_id': 'B', 'event_id': 'e1', 'type': 'ITEM_ADDED', 'sku': 'P2', 'qty': 5, 'timestamp': 1},
            {'order_id': 'B', 'event_id': 'e2', 'type': 'CANCELLED', 'timestamp': 2},
            {'order_id': 'B', 'event_id': 'e3', 'type': 'PAID', 'timestamp': 3}, 
            {'order_id': 'B', 'event_id': 'e4', 'type': 'ITEM_ADDED', 'sku': 'P3', 'qty': 1, 'timestamp': 4} 
        ]
        expected = [{'order_id': 'B', 'status': 'CANCELLED', 'items': {'P2': 5}}]
        self.assertEqual(summarize_orders(events), expected)

    def test_negative(self):
        events = [
            {'order_id': 'C', 'event_id': 'e1', 'type': 'ITEM_ADDED', 'sku': 'P1', 'qty': 2, 'timestamp': 1},
            {'order_id': 'C', 'event_id': 'e2', 'type': 'ITEM_REMOVED', 'sku': 'P1', 'qty': 5, 'timestamp': 2}
        ]
        expected = [{'order_id': 'C', 'status': 'CREATED', 'items': {}}]
        self.assertEqual(summarize_orders(events), expected)
